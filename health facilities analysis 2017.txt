Q12. Using CASEWHEN classify each county's dominant ownership as Government Heavy (more than 60% government), Mixed(40 to 60% government) or Private Heavy (less than 40% government).

WITH County_ownership AS (SELECT County,COUNT(*) AS total_facilities,
SUM(CASE WHEN Owner = 'Ministry of Health'
    THEN 1 ELSE 0 END)AS government,
SUM(CASE WHEN Owner != 'Ministry of Health'
     THEN 1 ELSE 0 END) AS nongorvernment,
ROUND(SUM(CASE WHEN Owner = 'Ministry of Health'
      THEN 1 ELSE 0 END) * 100.0/COUNT(*),1)
	  AS pct_government
FROM health
GROUP BY county
)
SELECT County,total_facilities,government,pct_government,CASE
WHEN pct_government > 60 THEN 'Government Heavy'
WHEN pct_government BETWEEN 40 AND 60 THEN 'Mixed'
ELSE 'Private Heavy'
END AS ownership_classification
FROM county_ownership
ORDER BY pct_government DESC;

Q17. Write a CTE that calculates average facilities per county nationally.Then show all counties below that average alongside their gap from the average.

WITH national_avg AS (
SELECT ROUND(AVG(county_total), 1) AS avg_facilities
FROM (SELECT County, COUNT(*) AS county_total
FROM health
GROUP BY county
)
),
county_totals AS (
SELECT County, COUNT(*) AS total_facilities
FROM health
GROUP BY county
)
SELECT c.County,c.total_facilities,n.avg_facilities AS national_average,
ROUND (n.avg_facilities - c.total_facilities, 1) AS gap_below_average
FROM county_totals c, national_avg n
WHERE c.total_facilities < n.avg_facilities
ORDER BY gap_below_average DESC;

Q20. Write a CTE that joins health_facilities to your Kenya_counties table to calculate facilities per 100,000 people per county.Then classify using CASE WHEN as Critical (below 10 per 100K),Low(10 to 25),Adequate (25 to 50), or Well Served(above 50)

WITH facility_counts AS (
SELECT County,COUNT(*) AS total_facilities
FROM health
GROUP BY county
),
county_access AS (
SELECT f.County,f.total_facilities,k.population,
ROUND(f.total_facilities * 100000.0 / k.population, 1)
AS facilities_per_100k
FROM facility_counts f
JOIN kenya_counties_clean k ON LOWER(f.County) = LOWER(k.county)
)
SELECT County,total_facilities,population,facilities_per_100k,
CASE
WHEN facilities_per_100k < 10 THEN 'Critical'
WHEN facilities_per_100k BETWEEN 10 AND 25 THEN 'Low'
WHEN facilities_per_100k BETWEEN 25 AND 50 THEN 'Adequate'
ELSE 'Well Served'
END AS access_classification
FROM county_access
ORDER BY facilities_per_100k ASC;
